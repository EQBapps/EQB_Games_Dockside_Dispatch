# Dockside Dispatch: R8 keep rules for release builds (Player Settings > Publishing > Minify Release, Custom Proguard File).
#
# R8 removes and renames Java/Kotlin code it can't see being used. C# reaches Java by class and method NAME over JNI
# (AndroidJavaClass, AndroidJavaObject, AndroidJavaProxy), which R8 can't see, so every Java API called from C# is kept
# here. Everything else (androidx, Play services internals, Kotlin, the ad networks, which ship their own consumer rules)
# is still optimized. Unity's own rules (com.unity3d.player, GameActivity) come from proguard-unity.txt.
#
# Crash stack traces: the mapping file is bundled in the AAB, so Play Console shows readable Java traces. Unity also
# saves a copy next to each build (Builds/Android/<build>_mapping.txt); keep it to retrace a logcat stack trace:
#   retrace <build>_mapping.txt trace.txt

-keepattributes Signature,InnerClasses,EnclosingMethod,*Annotation*,SourceFile,LineNumberTable

# --- Unity: player, Unity Ads / services (both called from C#) ---
-keep class com.unity3d.** { *; }
-dontwarn com.unity3d.**

-keepclassmembers class * implements android.os.Parcelable { public static final android.os.Parcelable$Creator *; }
-keep public class com.google.android.gms.ads.** { public *; }
-keepattributes JavascriptInterface

-dontwarn com.ironsource.mediationsdk.**
